cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD_HEADER_DIRS "${CMAKE_CURRENT_LIST_DIR}/boards")
set(PICO_BOARD waveshare_rp2350_usb_c_acceptance CACHE STRING "Board")
set(PICO_PLATFORM rp2350-arm-s CACHE STRING "Platform")
set(PICO_PIO_USB_PATH "${CMAKE_CURRENT_LIST_DIR}/vendor/pico-pio-usb")
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(usb_submcu_host_probe C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
if(NOT PICO_SDK_VERSION_STRING STREQUAL "2.2.0")
 message(FATAL_ERROR "Pinned SDK 2.2.0 required")
endif()
set(PROBE_UART_TX_PIN -1 CACHE STRING "Diagnostic TX only; -1 leaves all UART pins untouched")
if(NOT PROBE_UART_TX_PIN STREQUAL "-1" AND NOT PROBE_UART_TX_PIN STREQUAL "0")
 message(FATAL_ERROR "Only disabled or provisional GP0 UART0 TX is supported")
endif()
if(PROBE_UART_TX_PIN STREQUAL "0")
 set(PROBE_FW_VERSION "qr-uart-20261111.1")
 set(PROBE_PROGRAM_NAME "usb_submcu_qr_uart_probe")
else()
 set(PROBE_FW_VERSION "qr-probe-20261111.1")
 set(PROBE_PROGRAM_NAME "usb_submcu_qr_host_probe")
endif()
add_executable(usb_submcu_host_probe main.c newlib_malloc_lock.c)
target_include_directories(usb_submcu_host_probe PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_compile_definitions(usb_submcu_host_probe PRIVATE PROBE_UART_TX_PIN=${PROBE_UART_TX_PIN} PROBE_FW_VERSION="${PROBE_FW_VERSION}")
target_compile_options(usb_submcu_host_probe PRIVATE -Wall -Wextra)
set_source_files_properties(main.c newlib_malloc_lock.c PROPERTIES COMPILE_OPTIONS "-Werror")
target_link_libraries(usb_submcu_host_probe pico_stdlib pico_unique_id tinyusb_host tinyusb_pico_pio_usb hardware_uart)
pico_enable_stdio_usb(usb_submcu_host_probe 0)
pico_enable_stdio_uart(usb_submcu_host_probe 0)
pico_set_program_name(usb_submcu_host_probe "${PROBE_PROGRAM_NAME}")
pico_set_program_version(usb_submcu_host_probe "${PROBE_FW_VERSION}")
pico_add_extra_outputs(usb_submcu_host_probe)
