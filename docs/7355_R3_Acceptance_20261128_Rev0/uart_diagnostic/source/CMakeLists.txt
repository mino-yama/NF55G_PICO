cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD_HEADER_DIRS "${CMAKE_CURRENT_LIST_DIR}/boards")
set(PICO_BOARD waveshare_rp2350_usb_a_host_eval CACHE STRING "Known meter HOST")
set(PICO_PLATFORM rp2350-arm-s CACHE STRING "RP2350")
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(meter_uart_diagnostic C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(meter_uart_diagnostic main.c relay_host.c relay_wire.c)
target_include_directories(meter_uart_diagnostic PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_compile_options(meter_uart_diagnostic PRIVATE -Wall -Wextra -Werror)
target_link_libraries(meter_uart_diagnostic PRIVATE pico_stdlib pico_unique_id hardware_clocks hardware_uart)
pico_enable_stdio_uart(meter_uart_diagnostic 0)
pico_enable_stdio_usb(meter_uart_diagnostic 1)
pico_set_program_name(meter_uart_diagnostic "meter_uart_diagnostic")
pico_set_program_version(meter_uart_diagnostic "meter-host-relay-uartdiag-20261128.1")
pico_add_extra_outputs(meter_uart_diagnostic)
